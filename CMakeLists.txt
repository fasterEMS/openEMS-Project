
## usage: cmake -DCMAKE_INSTALL_PREFIX=<install/path> .
## use -DBUILD_APPCSXCAD=NO to not build QCSXCAD and AppCSXCAD

# Require CMake 3.4+ because we need "$<SEMICOLON>" in CMAKE_ARGS
cmake_minimum_required(VERSION 3.4...3.10)

PROJECT(openEMS-Project CXX)

set(OPENEMS_CMAKE_COMMON "-G${CMAKE_GENERATOR}")
message(STATUS "Using CMAKE_GENERATOR: " ${CMAKE_GENERATOR})

if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE "Release")
endif()
list(APPEND OPENEMS_CMAKE_COMMON "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}")
message(STATUS "Using CMAKE_BUILD_TYPE: " ${CMAKE_BUILD_TYPE})

if (CMAKE_TOOLCHAIN_FILE)
  message(STATUS "Using CMAKE_TOOLCHAIN_FILE: " ${CMAKE_TOOLCHAIN_FILE})
  list(APPEND OPENEMS_CMAKE_COMMON "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}")
endif()

if (CMAKE_INSTALL_PREFIX)
  message(STATUS "Using CMAKE_INSTALL_PREFIX: " ${CMAKE_INSTALL_PREFIX})
  list(APPEND OPENEMS_CMAKE_COMMON "-DCMAKE_INSTALL_PREFIX=${CMAKE_INSTALL_PREFIX}")
endif()

if (VCPKG_HOST_TRIPLET)
  message(STATUS "Using VCPKG_HOST_TRIPLET: " ${VCPKG_HOST_TRIPLET})
  list(APPEND OPENEMS_CMAKE_COMMON "-DVCPKG_HOST_TRIPLET=${VCPKG_HOST_TRIPLET}")
endif()

if (VCPKG_TARGET_TRIPLET)
  message(STATUS "Using VCPKG_TARGET_TRIPLET: " ${VCPKG_TARGET_TRIPLET})
  list(APPEND OPENEMS_CMAKE_COMMON "-DVCPKG_TARGET_TRIPLET=${VCPKG_TARGET_TRIPLET}")
endif()

if (VCPKG_INSTALL_OPTIONS)
  string(REPLACE ";" "$<SEMICOLON>" VCPKG_INSTALL_OPTIONS "${VCPKG_INSTALL_OPTIONS}")
  message(STATUS "Using VCPKG_INSTALL_OPTIONS: " ${VCPKG_INSTALL_OPTIONS})
  list(APPEND OPENEMS_CMAKE_COMMON "-DVCPKG_INSTALL_OPTIONS=${VCPKG_INSTALL_OPTIONS}")
endif()

if (NOT DEFINED BUILD_APPCSXCAD)
  set(BUILD_APPCSXCAD "YES")
endif()

message(STATUS "Build AppCSXCAD: " ${BUILD_APPCSXCAD})

IF(EXISTS ${PROJECT_SOURCE_DIR}/localConfig.cmake)
  message(STATUS "include local config" )
  include(${PROJECT_SOURCE_DIR}/localConfig.cmake)
ENDIF()

include(ExternalProject)

# build fparser
ExternalProject_Add( fparser
  SOURCE_DIR ${PROJECT_SOURCE_DIR}/fparser
  CMAKE_ARGS ${OPENEMS_CMAKE_COMMON}
  USES_TERMINAL_CONFIGURE true
)

# build CSXCAD
ExternalProject_Add( CSXCAD
  DEPENDS     fparser
  SOURCE_DIR  ${PROJECT_SOURCE_DIR}/CSXCAD
  CMAKE_ARGS
    ${OPENEMS_CMAKE_COMMON}
    -DFPARSER_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
  USES_TERMINAL_CONFIGURE true
)

message(STATUS "with MPI: ${WITH_MPI}" )
# build openEMS
ExternalProject_Add( openEMS
  DEPENDS     fparser CSXCAD
  SOURCE_DIR  ${PROJECT_SOURCE_DIR}/openEMS
  CMAKE_ARGS
    ${OPENEMS_CMAKE_COMMON}
    -DFPARSER_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
    -DCSXCAD_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
    -DWITH_MPI=${WITH_MPI}
  USES_TERMINAL_CONFIGURE true
)

if (${BUILD_APPCSXCAD})
    # build QCSXCAD
    ExternalProject_Add( QCSXCAD
    DEPENDS     CSXCAD
    SOURCE_DIR  ${PROJECT_SOURCE_DIR}/QCSXCAD
    CMAKE_ARGS
      ${OPENEMS_CMAKE_COMMON}
      -DCSXCAD_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
    USES_TERMINAL_CONFIGURE true
    )

    # build AppCSXCAD
    ExternalProject_Add( AppCSXCAD
    DEPENDS     QCSXCAD
    SOURCE_DIR  ${PROJECT_SOURCE_DIR}/AppCSXCAD
    CMAKE_ARGS
      ${OPENEMS_CMAKE_COMMON}
      -DCSXCAD_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
      -DQCSXCAD_ROOT_DIR=${CMAKE_INSTALL_PREFIX}
    USES_TERMINAL_CONFIGURE true
    )
endif()
